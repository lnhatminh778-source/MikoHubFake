-- Helper Functions
local function getRootPart()
    local char = LocalPlayer.Character
    if char then
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if hum and hum.Health > 0 and root then
            return root, hum
        end
    end
    return nil, nil
end

local function stopVelocity(root)
    if root then
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end
end

-- ================= ANTI LAG SYSTEM =================
local function EnableAntiLag()
    pcall(function()
        Lighting.GlobalShadows = false
        Lighting.FogEnd = 9e9
        for _, v in ipairs(Workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                v.Material = Enum.Material.SmoothPlastic
                v.Reflectance = 0
            elseif v:IsA("Decal") or v:IsA("Texture") then
                v:Destroy()
            elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") then
                v.Enabled = false
            end
        end
    end)
end

local function HopServer(reason)
    if isHopping then return end
    isHopping = true
    reason = reason or "Switching server..."
    
    WindUI:Notify({ Title = "Server Hop", Content = reason, Duration = 3 })
    
    -- Đảm bảo ép queueTeleport trước mỗi lần Hop
    SaveScriptForTeleport()

    task.spawn(function()
        local req = (syn and syn.request) or (http and http.request) or http_request or request
        local success = false
        
        if req then
            pcall(function()
                local ApiUrl = "https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?sortOrder=Desc&limit=100"
                local res = req({ Url = ApiUrl, Method = "GET" })
                if res and res.Body then
                    local decoded = HttpService:JSONDecode(res.Body)
                    if decoded and decoded.data then
                        for _, s in ipairs(decoded.data) do
                            if s.playing and s.maxPlayers and s.playing < (s.maxPlayers - 1) and s.id ~= game.JobId then
                                SaveScriptForTeleport()
                                TeleportService:TeleportToPlaceInstance(game.PlaceId, s.id, LocalPlayer)
                                success = true
                                break
                            end
                        end
                    end
                end
            end)
        end
        
        if not success then
            pcall(function()
                SaveScriptForTeleport()
                TeleportService:Teleport(game.PlaceId, LocalPlayer)
            end)
        end
        
        task.wait(8)
        if isHopping then
            isHopping = false
        end
    end)
end

-- Validation
local BAD_WORDS = {"revive", "help", "heal", "carry", "rescue", "drive", "ride", "enter", "vehicle", "shop", "buy", "talk"}

local function isValidItem(obj, prompt)
    if not obj or not obj.Parent then return false end
    
    if Players:GetPlayerFromCharacter(obj:FindFirstAncestorOfClass("Model")) then
        return false
    end
    
    if obj:IsA("Seat") or obj:IsA("VehicleSeat") then return false end
    
    local model = obj:IsA("Model") and obj or obj:FindFirstAncestorOfClass("Model")
    if model and not model:IsA("Tool") and model:FindFirstChildOfClass("Humanoid") then
        return false
    end
    
    if prompt then
        local actionText = (prompt.ActionText or ""):lower()
        local objectText = (prompt.ObjectText or ""):lower()
        for _, word in ipairs(BAD_WORDS) do
            if actionText:find(word, 1, true) or objectText:find(word, 1, true) then
                return false
            end
        end
    end
    return true
end

local function getMapItems()
    local items = {}
    for _, obj in ipairs(Workspace:GetDescendants()) do
        if obj:IsA("ProximityPrompt") then
            if obj.Parent and isValidItem(obj.Parent, obj) then
                table.insert(items, { Type = "Prompt", Target = obj.Parent, Prompt = obj })
            end
        elseif obj:IsA("TouchTransmitter") then
            if obj.Parent and obj.Parent:IsA("BasePart") and isValidItem(obj.Parent, nil) then
                table.insert(items, { Type = "Touch", Target = obj.Parent, Part = obj.Parent })
            end
        elseif obj:IsA("Tool") and obj.Parent == Workspace and isValidItem(obj, nil) then
            local handle = obj:FindFirstChild("Handle") or obj:FindFirstChildWhichIsA("BasePart")
            if handle then
                table.insert(items, { Type = "Tool", Target = obj, Part = handle })
            end
        end
    end
    return items
end

local function triggerPickup(char, itemData)
    if not char or not itemData or not itemData.Target or not itemData.Target.Parent then return end
    
    if itemData.Prompt and fireproximityprompt then
        local prompt = itemData.Prompt
        prompt.HoldDuration = 0
        prompt.RequiresLineOfSight = false
        prompt.MaxActivationDistance = 9999
        prompt.Enabled = true
        fireproximityprompt(prompt, 0)
        fireproximityprompt(prompt, 1)
    end
    
    local targetPart = itemData.Part or (itemData.Target:IsA("BasePart") and itemData.Target or itemData.Target:FindFirstChildWhichIsA("BasePart", true))
    if targetPart and firetouchinterest then
        for _, part in ipairs(char:GetChildren()) do
            if part:IsA("BasePart") then
                firetouchinterest(part, targetPart, 0)
                firetouchinterest(part, targetPart, 1)
            end
        end
    end
end

-- Auto Farm Main Loop
task.spawn(function()
    task.wait(1.5)
    local emptyCheckCounter = 0
    
    while true do
        task.wait(ScanDelay)
        
        if Config.AutoFarmEnabled and not isHopping then
            local root, hum = getRootPart()
            
            if root and hum then
                local items = getMapItems()
                
                if #items > 0 then
                    emptyCheckCounter = 0
                    local pickedAny = false
                    
                    for _, itemData in ipairs(items) do
                        if not Config.AutoFarmEnabled or isHopping or hum.Health <= 0 then break end
                        
                        local target = itemData.Target
                        if target and target.Parent then
                            local targetCFrame = itemData.Part and itemData.Part.CFrame 
                                or (target:IsA("BasePart") and target.CFrame) 
                                or (target:IsA("Model") or target:IsA("Tool")) and target:GetPivot()
                            
                            if targetCFrame then
                                root.Anchored = false
                                root.CFrame = targetCFrame * CFrame.new(0, 0.5, 0)
                                stopVelocity(root)
                                
                                triggerPickup(LocalPlayer.Character, itemData)
                                pickedAny = true
                                
                                task.wait(Config.PickupDelay)
                            end
                        end
                    end
                    
                    local remainingItems = getMapItems()
                    if (#remainingItems == 0 or not pickedAny) and Config.AutoServerHopItem then
                        HopServer("Finished collecting items! Hopping server...")
                    end
                else
                    emptyCheckCounter = emptyCheckCounter + 1
                    if Config.AutoServerHopItem and emptyCheckCounter >= 2 then
                        HopServer("No items found on map! Hopping server...")
                    end
                end
            end
        end
    end
end)
